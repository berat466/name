cmake_minimum_required(VERSION 3.16)
project(V81_LOCAL_BENCHMARK CXX)
# V81 (ve karsilastirma icin V80) kaynaklarini DEGISTIRMEDEN derler. Uyumluluk: compat/th_compat.h (-include ile on-eklenir).
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)   # GNU uzantilari (bits/stdc++.h, __builtin_*)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Kaynak dizini: paket icinde src/, depo icinde ../cpp
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/src/tek_hamle_v81.cpp")
  set(TH_SRC_DIR "${CMAKE_CURRENT_LIST_DIR}/src")
else()
  set(TH_SRC_DIR "${CMAKE_CURRENT_LIST_DIR}/../cpp")
endif()
message(STATUS "Kaynak dizini: ${TH_SRC_DIR}")

set(TH_ISA "auto" CACHE STRING "auto | avx512 | avx2 | both")
set(TH_STD_BLAS ON CACHE BOOL "Standart OpenBLAS adlarini (cblas_*, dpotrf_, dsyev_) kullan; OFF: scipy on-ekli OpenBLAS (scipy_cblas_*)")
set(TH_BLAS_LIB "" CACHE STRING "OpenBLAS kutuphane yolu (bos: otomatik bul)")

find_package(OpenMP REQUIRED)
if(TH_BLAS_LIB)
  set(TH_BLAS "${TH_BLAS_LIB}")
else()
  find_library(TH_BLAS NAMES openblas libopenblas REQUIRED)
endif()
message(STATUS "OpenBLAS: ${TH_BLAS}")
include(CheckLibraryExists)
check_library_exists("${TH_BLAS}" openblas_set_num_threads_local "" TH_HAVE_OB_LOCAL)
message(STATUS "openblas_set_num_threads_local mevcut: ${TH_HAVE_OB_LOCAL}  (yoksa th_compat.h bos taklit tanimlar)")

if(NOT (CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang"))
  message(FATAL_ERROR "Desteklenen derleyici: GCC (MSYS2/MinGW-w64 UCRT64) veya Clang. MSVC DESTEKLENMIYOR (kaynak GCC uzantilari kullanir: bits/stdc++.h, __attribute__, __builtin_*; OpenMP 2.0 yetersiz).")
endif()

# CPU ozelligi tespiti (derleme makinesinde) -- yalniz 'auto' icin
set(_has512 FALSE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|x86")
  file(WRITE "${CMAKE_BINARY_DIR}/cpuprobe.cpp" "#include <cstdio>\nint main(){__builtin_cpu_init();printf(\"%d\",(int)(__builtin_cpu_supports(\"avx512f\")&&__builtin_cpu_supports(\"avx512dq\")&&__builtin_cpu_supports(\"avx512bw\")&&__builtin_cpu_supports(\"avx512vl\")&&__builtin_cpu_supports(\"avx512cd\")));return 0;}\n")
  try_run(_run_rc _compile_ok "${CMAKE_BINARY_DIR}" "${CMAKE_BINARY_DIR}/cpuprobe.cpp" RUN_OUTPUT_VARIABLE _probe_out)
  if(_compile_ok AND _run_rc EQUAL 0 AND _probe_out STREQUAL "1")
    set(_has512 TRUE)
  endif()
endif()
message(STATUS "Bu CPU AVX-512 (F/CD/BW/DQ/VL) destekliyor mu: ${_has512}")

set(_want512 FALSE)
set(_want2 FALSE)
if(TH_ISA STREQUAL "avx512")
  set(_want512 TRUE)
elseif(TH_ISA STREQUAL "avx2")
  set(_want2 TRUE)
elseif(TH_ISA STREQUAL "both")
  set(_want512 TRUE)
  set(_want2 TRUE)
else()   # auto
  if(_has512)
    set(_want512 TRUE)
  endif()
  set(_want2 TRUE)
endif()

function(th_target name src isaflags isa)
  add_executable(${name} "${TH_SRC_DIR}/${src}")
  target_include_directories(${name} PRIVATE "${TH_SRC_DIR}" "${CMAKE_CURRENT_LIST_DIR}/compat")
  if(WIN32)
    target_include_directories(${name} BEFORE PRIVATE "${CMAKE_CURRENT_LIST_DIR}/compat/win")
    target_link_libraries(${name} PRIVATE psapi)
  endif()
  target_compile_options(${name} PRIVATE -O3 -fno-math-errno ${isaflags} -include "${CMAKE_CURRENT_LIST_DIR}/compat/th_compat.h" -w)
  if(TH_STD_BLAS)
    target_compile_definitions(${name} PRIVATE TH_STD_BLAS=1)
  endif()
  if(NOT TH_HAVE_OB_LOCAL)
    target_compile_definitions(${name} PRIVATE TH_SHIM_OB_LOCAL=1)
  endif()
  target_link_libraries(${name} PRIVATE OpenMP::OpenMP_CXX "${TH_BLAS}")
  set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endfunction()

set(_f512 -mavx512f -mavx512cd -mavx512bw -mavx512dq -mavx512vl -mavx2 -mfma -mbmi2)
set(_f2 -mavx2 -mfma -mbmi2)
if(_want512)
  th_target(tek_hamle_v81_avx512 tek_hamle_v81.cpp "${_f512}" avx512)
  th_target(tek_hamle_v80_avx512 tek_hamle_v80.cpp "${_f512}" avx512)
endif()
if(_want2)
  th_target(tek_hamle_v81_avx2 tek_hamle_v81.cpp "${_f2}" avx2)
  th_target(tek_hamle_v80_avx2 tek_hamle_v80.cpp "${_f2}" avx2)
endif()

  file(WRITE "${CMAKE_BINARY_DIR}/bin/ISA_SECILEN.txt" "avx512\n")
else()
  file(WRITE "${CMAKE_BINARY_DIR}/bin/ISA_SECILEN.txt" "avx2\n")
endif()
